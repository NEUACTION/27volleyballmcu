cmake_minimum_required(VERSION 3.8)
project(action_planner)

# 启用常规编译诊断，使用 C++17 调用 planner 在线求值接口。
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(base_interfaces REQUIRED)
find_package(planner REQUIRED)
add_executable(action_planner_node src/action_planner_node.cpp)
target_compile_features(action_planner_node PUBLIC cxx_std_17)
ament_target_dependencies(action_planner_node rclcpp base_interfaces)
target_link_libraries(action_planner_node planner::planner)
install(TARGETS action_planner_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
